-"use server";

import { cookies } from "next/headers";
import { redirect } from "next/navigation";

const BACKEND_URL = process.env.BACKEND_URL ?? "http://localhost:8080";

export interface OnboardingActionState {
  success: boolean;
  message: string;
}

/**
 * Server Action backing the onboarding form.
 * Uses a direct fetch() instead of apiFetch — bypasses whatever apiFetch's
 * own header/cookie logic was doing that caused this POST to arrive at
 * Spring as a GET. This call is fully explicit about method/headers/body.
 *
 * Expects FormData with:
 *   - userId
 *   - roleIds (JSON-stringified string[])
 *   - technologyIds (JSON-stringified string[])
 *   - intentId
 */
export async function submitOnboardingAction(
  prevState: OnboardingActionState | null,
  formData: FormData,
): Promise<OnboardingActionState> {
  const userId = formData.get("userId") as string;
  const intentId = formData.get("intentId") as string;
  const roleIds = JSON.parse(formData.get("roleIds") as string) as string[];
  const technologyIds = JSON.parse(
    formData.get("technologyIds") as string,
  ) as string[];

  if (
    !userId ||
    !intentId ||
    roleIds.length === 0 ||
    technologyIds.length === 0
  ) {
    return {
      success: false,
      message: "Please complete all sections before continuing.",
    };
  }

  const cookieStore = await cookies();
  const token = cookieStore.get("auth_token")?.value;

  if (!token) {
    return {
      success: false,
      message: "You're not logged in. Please log in again.",
    };
  }

  let res: Response;
  try {
    res = await fetch(`${BACKEND_URL}/onboarding`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Authorization: token,
      },
      body: JSON.stringify({ userId, roleIds, technologyIds, intentId }),
      cache: "no-store",
    });
  } catch (err) {
    return {
      success: false,
      message: "Could not reach the server. Please try again.",
    };
  }

  if (!res.ok) {
    const body = await res.text().catch(() => "");
    return {
      success: false,
      message: body || `Request failed with status ${res.status}`,
    };
  }

  redirect(`/matches/${userId}`);
}

async function getOnboardingCompleted(
  userId: string,
  token: string,
): Promise<boolean> {
  try {
    const res = await fetch(`${BACKEND_URL}/users/${userId}/status`, {
      headers: { Authorization: token },
      cache: "no-store",
    });
    if (!res.ok) return false;
    const data = await res.json();
    return Boolean(data.onboardingCompleted);
  } catch {
    return false;
  }
}

5. Sample hook
import * as React from "react"

const MOBILE_BREAKPOINT = 768

export function useIsMobile() {
  const [isMobile, setIsMobile] = React.useState<boolean | undefined>(undefined)

  React.useEffect(() => {
    const mql = window.matchMedia(`(max-width: ${MOBILE_BREAKPOINT - 1}px)`)
    const onChange = () => {
      setIsMobile(window.innerWidth < MOBILE_BREAKPOINT)
    }
    mql.addEventListener("change", onChange)
    setIsMobile(window.innerWidth < MOBILE_BREAKPOINT)
    return () => mql.removeEventListener("change", onChange)
  }, [])

  return !!isMobile
}

7. Sample auth.ts

// lib/auth.ts
import { cookies } from "next/headers";

export async function getCurrentUserId(): Promise<string | null> {
  try {
    const cookieStore = await cookies();
    const token = cookieStore.get("auth_token")?.value;

    if (!token) return null;

    // Decode JWT to get userId
    const payload = JSON.parse(atob(token.split(".")[1]));
    return payload.userId || payload.sub || null;
  } catch (error) {
    return null;
  }
}

// lib/api/onboarding.ts
import { apiFetch } from "@/lib/api-client";
import type {
  Role,
  Technology,
  Intent,
  OnboardingPayload,
  OnboardingResponse,
} from "@/lib/schema";

/**
 * GET /roles
 * Mirrors RoleController#getAllRoles
 */
export async function fetchRoles(): Promise<Role[]> {
  return apiFetch<Role[]>("/roles");
}

/**
 * GET /technologies
 * Mirrors a TechnologyController#getAllTechnologies (same shape as Role/Intent controllers).
 * If your backend route differs, update the path below only — callers don't change.
 */
export async function fetchTechnologies(): Promise<Technology[]> {
  return apiFetch<Technology[]>("/technologies");
}

/**
 * GET /intents
 * Mirrors IntentController#getAllIntents
 */
export async function fetchIntents(): Promise<Intent[]> {
  return apiFetch<Intent[]>("/intents");
}

/**
 * POST /onboarding
 * Mirrors OnboardingController — submits the user's selections, persists them,
 * and the backend triggers MatchService#createMatchesForUser, returning the
 * created matches.
 *
 * This is a mutation, so it intentionally skips the GET cache settings used above.
 */

export async function submitOnboarding(
  payload: OnboardingPayload,
): Promise<OnboardingResponse> {
  let token: string | undefined;

  try {
    const { cookies } = await import("next/headers");
    const cookieStore = await cookies();
    token = cookieStore.get("auth_token")?.value;

    console.log("Retrieved auth token from cookies:", token);
  } catch (error) {
    // Client-side fallback
    if (typeof document !== "undefined") {
      token = document.cookie
        .split("; ")
        .find((row) => row.startsWith("auth_token="))
        ?.split("=")[1];
    }
  }

  if (!token) {
    console.error("No auth token found");
    throw new Error("No auth token found");
  }

  return apiFetch<OnboardingResponse>("/onboarding", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: token,
      credentials: "include",
    },
    body: JSON.stringify(payload),
    cache: "no-store",
  });
}
